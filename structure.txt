--------------------------------------------------------------------------------
1. PROJECT FILE STRUCTURE
--------------------------------------------------------------------------------

URVI-MAIN/
│
├── .vscode/
│   └── settings.json            # VS Code editor configuration
│
├── activities/
│   └── activities.html          # User eco-activities tracking page
│
├── assets/
│   ├── logo.png                 # Main URVI logo image asset
│   └── README.txt               # Assets documentation file
│
├── community/
│   ├── community.css            # Styling for the Community feed page
│   ├── community.html           # Main Community feed page (posts, widgets, user search)
│   └── community.js             # Logic for Community feed, post creation, likes, comments & notifications
│
├── components/
│   ├── layout.js                # Common layout elements / component builder
│   ├── notifications.js         # Header notification bell widget logic
│   ├── pwa.js                   # Progressive Web App service worker registration & install prompts
│   ├── translator.css           # Styling for language translator widget
│   ├── translator.js            # Translation & language selector logic
│   └── vineDecoration.html      # Vine decorative SVG/HTML component
│
├── impact/
│   └── impact.html              # Impact metrics dashboard page (trees planted, carbon saved)
│
├── index/
│   ├── libs/
│   │   ├── bootstrap-icons.min.css # Icon font CSS stylesheet
│   │   ├── bootstrap-icons.woff     # Web font asset (WOFF)
│   │   └── bootstrap-icons.woff2    # Web font asset (WOFF2)
│   ├── index.css                # CSS styles specific to homepage
│   ├── index.html               # Main homepage layout component
│   └── index.js                 # Homepage interactive features and logic
│
├── logins/
│   ├── login.html               # User login form & auth logic
│   ├── logins.css               # Styling for login and signup pages
│   └── signup.html              # User signup registration form & support request modal
│
├── notifications/
│   ├── notifications.css        # Styling for full notifications page
│   ├── notifications.html       # Notifications center page layout
│   └── notifications.js         # Notifications filtering, rendering & mark-as-read logic
│
├── profile/
│   ├── profile.css              # Styling for user profile pages
│   ├── profile.html             # Logged-in user's own profile page
│   ├── profile.js               # Logic for managing own profile, editing bio/photo & user posts
│   ├── user-profile.html        # Public profile page for viewing other users
│   └── user-profile.js          # Logic for viewing public profiles & follow/unfollow functionality
│
├── cloudinary.js                # Cloudinary integration utility for image/video uploads
├── config.js                    # Firebase app initialization & database configuration export
├── index.html                   # Root application entry point
├── logo.png                     # Application logo image file
├── manifest.json                # PWA web app manifest metadata
└── sw.js                        # Service Worker for offline caching & PWA support
